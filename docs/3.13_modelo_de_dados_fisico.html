6. Mapeamento para Cloud SQL / BigQuery (DDL Projetado)

Caso a migração para persistência gerenciada seja realizada (conforme item 1.1 do parecer UTIC), os schemas acima podem ser mapeados para Cloud SQL (PostgreSQL) ou BigQuery. Abaixo, o DDL projetado para as entidades principais:

-- Tabela principal: emendas enriquecidas
CREATE TABLE emendas (
    codigo_emenda     VARCHAR(20) PRIMARY KEY,
    ano               INTEGER NOT NULL,
    tipo_emenda       VARCHAR(80),
    codigo_autor      VARCHAR(20),
    nome_autor        VARCHAR(200),
    numero_emenda     VARCHAR(20),
    localidade_aplicacao TEXT,
    cod_ibge_municipio VARCHAR(7),
    municipio         VARCHAR(100),
    cod_ibge_uf       VARCHAR(2),
    uf_destino        VARCHAR(2),
    regiao            VARCHAR(20),
    cod_funcao        VARCHAR(5),
    funcao            VARCHAR(100),
    cod_subfuncao     VARCHAR(5),
    subfuncao         VARCHAR(100),
    cod_programa      VARCHAR(10),
    programa          VARCHAR(200),
    cod_acao          VARCHAR(10),
    acao              VARCHAR(200),
    cod_plano_orcamentario VARCHAR(10),
    plano_orcamentario VARCHAR(200),
    valor_empenhado   NUMERIC(18,2),
    valor_liquidado   NUMERIC(18,2),
    valor_pago        NUMERIC(18,2),
    valor_rp_inscrito NUMERIC(18,2),
    valor_rp_cancelado NUMERIC(18,2),
    valor_rp_pago     NUMERIC(18,2),
    localidade_tipo   VARCHAR(20),
    partido           VARCHAR(20),
    uf_autor          VARCHAR(2),
    legislatura       INTEGER,
    match_confianca   VARCHAR(20)
);

CREATE INDEX idx_emendas_ano ON emendas(ano);
CREATE INDEX idx_emendas_uf ON emendas(uf_destino);
CREATE INDEX idx_emendas_autor ON emendas(nome_autor);
CREATE INDEX idx_emendas_partido ON emendas(partido);

-- Validação cruzada CGU x Tesouro
CREATE TABLE quality_flags (
    uf              VARCHAR(2),
    ano             INTEGER,
    valor_portal    NUMERIC(18,2),
    valor_tesouro   NUMERIC(18,2),
    divergencia_pct NUMERIC(6,2),
    status          VARCHAR(10),
    PRIMARY KEY (uf, ano)
);

-- Stakeholders (parlamentares)
CREATE TABLE parlamentares (
    id        INTEGER PRIMARY KEY,
    nome      VARCHAR(200) NOT NULL,
    partido   VARCHAR(20),
    uf        VARCHAR(2),
    casa      VARCHAR(10),
    foto_url  TEXT,
    email     VARCHAR(200)
);

CREATE TABLE parlamentar_frentes (
    parlamentar_id INTEGER REFERENCES parlamentares(id),
    frente_titulo  TEXT,
    PRIMARY KEY (parlamentar_id, frente_titulo)
);

CREATE TABLE parlamentar_comissoes (
    parlamentar_id INTEGER REFERENCES parlamentares(id),
    comissao_sigla VARCHAR(20),
    papel          VARCHAR(20),  -- Titular, Suplente, Presidente
    PRIMARY KEY (parlamentar_id, comissao_sigla)
);

-- Contratações PNCP (agregado)
CREATE TABLE contratacoes_pncp (
    uf         VARCHAR(2),
    ano        INTEGER,
    nome_uf    VARCHAR(50),
    regiao     VARCHAR(2),
    estimado   NUMERIC(18,2),
    homologado NUMERIC(18,2),
    qtd        INTEGER,
    PRIMARY KEY (uf, ano)
);

-- Séries econômicas BCB
CREATE TABLE series_bcb (
    serie_id   VARCHAR(10),
    data_obs   DATE,
    valor      NUMERIC(18,4),
    PRIMARY KEY (serie_id, data_obs)
);

-- Notícias (reforma tributária)
CREATE TABLE noticias_reforma (
    link       TEXT PRIMARY KEY,
    titulo     TEXT,
    fonte      VARCHAR(200),
    pub_date   TIMESTAMP
);

-- Execução orçamentária estadual (SICONFI/RREO)
CREATE TABLE exec_orcamentaria (
    uf         VARCHAR(2),
    ano        INTEGER,
    nome_uf    VARCHAR(50),
    regiao     VARCHAR(2),
    empenhado  NUMERIC(18,2),
    liquidado  NUMERIC(18,2),
    pago       NUMERIC(18,2),
    PRIMARY KEY (uf, ano)
);

-- PDDE — Programa Dinheiro Direto na Escola
CREATE TABLE pdde (
    uf         VARCHAR(2),
    ano        INTEGER,
    custeio    NUMERIC(18,2),
    capital    NUMERIC(18,2),
    total      NUMERIC(18,2),
    escolas    INTEGER,
    municipios INTEGER,
    alunos     INTEGER,
    PRIMARY KEY (uf, ano)
);

-- PNAE — Programa Nacional de Alimentação Escolar
CREATE TABLE pnae (
    uf         VARCHAR(2),
    ano        INTEGER,
    valor      NUMERIC(18,2),
    alunos     INTEGER,
    escolas    INTEGER,
    municipios INTEGER,
    PRIMARY KEY (uf, ano)
);

-- Candidatos 2026 (cruzamento TSE x APIs legislativas)
CREATE TABLE candidatos_2026 (
    parlamentar_nome VARCHAR(200) PRIMARY KEY,
    cargo            VARCHAR(30),
    uf               VARCHAR(2),
    eleito           BOOLEAN,
    nome_urna        VARCHAR(200)
);

-- Pesquisas eleitorais
CREATE TABLE pesquisas_eleitorais (
    id               VARCHAR(50) PRIMARY KEY,
    instituto        VARCHAR(100),
    data_publicacao  DATE,
    cargo            VARCHAR(30),
    uf               VARCHAR(5),
    amostra          INTEGER,
    margem           NUMERIC(4,2),
    metodologia      VARCHAR(20)
);
